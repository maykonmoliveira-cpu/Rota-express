Tabela entregas CREATE TABLE entregas ( id SERIAL PRIMARY KEY, user_id UUID NOT NULL REFERENCES auth.users(id), endereco TEXT NOT NULL, status TEXT DEFAULT 'pendente', latitude NUMERIC, longitude NUMERIC, created_at TIMESTAMP DEFAULT NOW() ); -- 2. Tabela rotas CREATE TABLE rotas ( id SERIAL PRIMARY KEY, user_id UUID NOT NULL REFERENCES auth.users(id), nome TEXT, created_at TIMESTAMP DEFAULT NOW() ); -- 3. Tabela paradas_rota CREATE TABLE paradas_rota ( id SERIAL PRIMARY KEY, rota_id INT NOT NULL REFERENCES rotas(id), entrega_id INT REFERENCES entregas(id), ordem INT ); -- 4. Suporte tickets CREATE TABLE support_tickets ( id SERIAL PRIMARY KEY, user_id UUID NOT NULL REFERENCES auth.users(id), subject TEXT NOT NULL, category TEXT, status TEXT DEFAULT 'aberto', created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW() ); -- 5. Mensagens suporte CREATE TABLE support_messages ( id SERIAL PRIMARY KEY, ticket_id INT NOT NULL REFERENCES support_tickets(id), sender_id UUID, sender_role TEXT CHECK (sender_role IN ('cliente','suporte')), body TEXT, created_at TIMESTAMP DEFAULT NOW() ); -- 6. Administradores CREATE TABLE support_admins ( id SERIAL PRIMARY KEY, user_id UUID PRIMARY KEY REFERENCES auth.users(id) );
