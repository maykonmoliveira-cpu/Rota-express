// FUNÇÃO GERAL PARA CANVAS
function configurarCanvas(idCanvas, limparFn, salvarFn){
  const canvas = document.getElementById(idCanvas);
  const ctx = canvas.getContext("2d");
  let desenhando = false;

  function ajustar(){
    const ratio = window.devicePixelRatio || 1;
    const w = canvas.clientWidth;
    const h = canvas.clientHeight;
    canvas.width = w * ratio;
    canvas.height = h * ratio;
    ctx.scale(ratio,ratio);
  }
  ajustar();
  window.addEventListener("resize",ajustar);

  function pegarPosicao(e){
    const rect = canvas.getBoundingClientRect();
    const pos = e.touches ? e.touches[0] : e;
    return {x:pos.clientX‑rect.left, y:pos.clientY‑rect.top};
  }

  function iniciarDesenho(e){
    desenhando = true;
    const p = pegarPosicao(e);
    ctx.beginPath();
    ctx.moveTo(p.x,p.y);
    ctx.lineWidth=2;
    ctx.lineCap="round";
  }
  function desenhar(e){
    if(!desenhando) return;
    const p = pegarPosicao(e);
    ctx.lineTo(p.x,p.y);
    ctx.stroke();
  }
  function parar(){desenhando=false;}

  canvas.addEventListener("mousedown",iniciarDesenho);
  canvas.addEventListener("touchstart",iniciarDesenho,{passive:false});
  canvas.addEventListener("mousemove",desenhar);
  canvas.addEventListener("touchmove",desenhar,{passive:false});
  canvas.addEventListener("mouseup",parar);
  canvas.addEventListener("touchend",parar);

  return {canvas,ctx};
}

// CLIENTE
const cCliente = configurarCanvas("canvasCliente");
function limparCanvasCliente(){cCliente.ctx.clearRect(0,0,cCliente.canvas.width,cCliente.canvas.height);}
function salvarCanvasCliente(){document.getElementById("imgAssinaturaCliente").value = cCliente.canvas.toDataURL("image/png");alert("✅ Assinatura do cliente salva!");}
